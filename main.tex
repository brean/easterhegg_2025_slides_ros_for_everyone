\begin{frame}{ROS 2 vs ROS 1}{ROS 1 is still everywhere}
%  Confusion \#1: 
\question{What is catkin? Why doesn't the roslaunch command exist?}\\

\begin{itemize}
  \item ROS 1 had ros... commands (without space), ROS 2 has the \textbf{ros2} command
  \item ROS 1 uses catkin ros 2 uses \textbf{colcon}
\end{itemize}

ROS 1 has its EOL in \textbf{May 2025}, the ros Wiki \url{https://wiki.ros.org/noetic/} is still linked everywhere!

The docs for jazzy are at \url{https://docs.ros.org/en/jazzy/}
\end{frame}

\begin{frame}{Nodes}{What is a ROS Node?}
  % Confusion \#3: 
  \question{Is a ROS node different from any other program?}\\
  No, a \textbf{node} is a small program that should be responsible for a single, module purpose.
  It can be part of a \textbf{ROS package} but does not have to be.
  \begin{figure}[tbh!]
      \centering
    \includegraphics[width=.75\textwidth]{./figures/nodes.pdf}
  \end{figure}
  Nodes are normally written in Python or C++.
  % https://docs.ros.org/en/foxy/Tutorials/Understanding-ROS2-Nodes.html
\end{frame}

\begin{frame}{Communication}{ROS Topics}
    Nodes can use \textbf{topics} to communicate. Use topics to send fast and repeating data \\\small (for example sensor readings). \normalsize

    \centering

    \includegraphics[width=.5\textwidth]{./figures/topic.pdf}

    \begin{itemize}
      \item the teleop-node publishes a TwistStamped message on the \texttt{cmd\_vel} topic.
      \item the motor controller-node subscribes to the \texttt{cmd\_vel} topic and receives it.
      \item other nodes can also talk or listen to \texttt{cmd\_vel}!
    \end{itemize}
\end{frame}

\begin{frame}{Communication}{ROS Services}

    \centering

    \includegraphics[width=.5\textwidth]{./figures/services.pdf}

    \begin{itemize}
      \item Topic-Problem: You don't know if your message got received.
      \item ROS services force you to send feedback with your message!
      \item Multiple nodes can call the same service, but only one node can provide one!
    \end{itemize}

    \question{Can we use messages for topics and services?}\\
    Yes, if you look at the implementation its all just \textbf{Messages}
\end{frame}

\begin{frame}{Communication}{ROS server}
  % Confusion \#6:
  \question{How do ROS nodes connect?}\\
  In ROS 1 we had a \textbf{rosmaster} you needed to start manually\\
  In ROS 2 we have the \textbf{ros2 daemon}.\\
  A Data Distribution Service (DDS) in the background makes sure everyone can connect.

\end{frame}

\begin{frame}{Robotics for Hackers}{Robots are not very secure.}
  \begin{itemize}
    \item Some manufacturer update software very irregular
    \item Very complex software stacks
    \item Software need to be extra safe
    \item Fluctuation of people/companies
    \item Your shiny new robot might come with a backdoor\\ (e.g. \url{https://think-awesome.com/download_unitree_report} by Bin4ry)
  \end{itemize}
  $\rightarrow$ Robots often get their own networks without internet.\\
  % \small{(anecdote: Time synchronization.)}
\end{frame}
